[[styles]]
== Styling Cells

=== Data Format

[source,kotlin]
----
val builder = ExcelBuilder()

builder.config {
    dataFormat("money", "# ##0.00")
    style("money-style") { cellStyle ->
        cellStyle.dataFormat = dataFormat("money")
    }

    style("number-style") { cellStyle ->
        cellStyle.dataFormat = dataFormat("number", "#0")
    }
}

val workbook = builder.build {
    sheet {
        row {
            cell(style = "money-style") { 1000.00 }
            cell(style = "number-style") { 1000 }
        }
    }
}
----

=== Fonts

Fonts are configured through Apache POI API:
https://poi.apache.org/apidocs/dev/org/apache/poi/xssf/usermodel/XSSFFont.html[XSSFFont]

[source,kotlin]
----
val builder = ExcelBuilder()

builder.config {
    font("configured-font") { font ->
        font.fontName = "Arial"
    }

    style("style-with-font") { cellStyle ->
        cellStyle.setFont(font("configured-font"))
    }
}
----

=== Styles

Styles are configured once and then applied to rows/cells.

[source,kotlin]
----
val builder = ExcelBuilder()

builder.config {
    style("centered") { cellStyle ->
        cellStyle.alignment = HorizontalAlignment.CENTER
        cellStyle.borderBottom = BorderStyle.DASH_DOT
        cellStyle.wrapText = true
    }
}
----
