== Examples

NOTE: Examples below show equivalent scenarios to Apache POI quick guide, implemented via this Kotlin DSL.

=== Create workbook, sheet and cells

[source,kotlin]
----
val wb = ExcelBuilder().build {
    sheet(name = "new sheet") {
        row {
            cell { 1 }
            cell { 1.2 }
            cell { "This is a string" }
            cell { true }
        }
    }
}

FileOutputStream("workbook.xlsx").use { wb.write(it) }
----

=== Date and Calendar cells

[source,kotlin]
----
val builder = ExcelBuilder()

builder.config {
    dataFormat("date-time", "m/d/yy h:mm")
    style("date-time-style") { cellStyle ->
        cellStyle.dataFormat = dataFormat("date-time")
    }
}

val workbook = builder.build {
    sheet(name = "new sheet") {
        row(style = "date-time-style") {
            cell { Date() }
            cell(style = "date-time-style") { Calendar.getInstance() }
        }
    }
}
----

=== Merge cells

[source,kotlin]
----
val workbook = ExcelBuilder().build {
    sheet(name = "merge") {
        row {
            cell(colspan = 2) { "This is a test of merging" }
        }
    }
}
----

=== Skip rows and cells

[source,kotlin]
----
val workbook = ExcelBuilder().build {
    sheet {
        row {
            cell { "A" }
            skipCell()
            cell { "B" }
            skipCell(2)
            cell { "C" }
        }

        skipRow()

        row {
            cell { "D" }
        }
    }
}
----

=== Fonts, style and data format

[source,kotlin]
----
val builder = ExcelBuilder()

builder.config {
    font("default-font") { f ->
        f.fontName = "Arial"
    }

    dataFormat("money", "# ##0.00")

    style("header") { s ->
        s.setFont(font("default-font"))
        s.alignment = HorizontalAlignment.CENTER
        s.borderBottom = BorderStyle.DOUBLE
    }

    style("money") { s ->
        s.dataFormat = dataFormat("money")
    }
}

val workbook = builder.build {
    sheet(name = "report") {
        row(style = "header") {
            cell { "Name" }
            cell { "Amount" }
        }
        row {
            cell { "A" }
            cell(style = "money") { 1200.50 }
        }
    }
}
----
