[[row]]
== Rows

=== Creating Rows

[source,kotlin]
----
val workbook = ExcelBuilder().build {
    sheet {
        row {
        }

        row()

        row(height = 10) {
        }
    }
}
----

=== Skipping Rows

[source,kotlin]
----
val workbook = ExcelBuilder().build {
    sheet {
        row {
            cell { "A" }
        }

        skipRow()

        row {
            cell { "B" }
        }

        skipRow(2)

        row {
            cell { "C" }
        }
    }
}
----

=== Styling Rows

[source,kotlin]
----
val workbook = ExcelBuilder().build {
    sheet {
        row(style = "row-style") {
        }
    }
}
----

=== Default Row Height

[source,kotlin]
----
val workbook = ExcelBuilder().build {
    sheet {
        row(height = ExcelBuilder.defaultRowHeight * 2) {
        }
    }
}
----
